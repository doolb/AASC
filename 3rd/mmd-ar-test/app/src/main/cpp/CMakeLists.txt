cmake_minimum_required(VERSION 3.22)
project(aasc_mmd_slam LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../" ABSOLUTE)
set(ORB "${PROJECT_ROOT}/build/third_party/orb-slam3")
set(DEPS "${PROJECT_ROOT}/build/third_party/mmd-ar-slam")
set(OpenCV_DIR "${DEPS}/OpenCV-android-sdk/sdk/native/jni")
set(OpenCV_STATIC ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc features2d calib3d imgcodecs)
set(EIGEN "${DEPS}/eigen-3.4.0")
set(BOOST "${DEPS}/boost_1_85_0")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/eigen3")
file(CREATE_LINK "${EIGEN}/Eigen" "${CMAKE_CURRENT_BINARY_DIR}/eigen3/Eigen" SYMBOLIC)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/compat" "${ORB}" "${ORB}/include"
    "${ORB}/include/CameraModels" "${ORB}/Thirdparty/Sophus" "${EIGEN}" "${BOOST}" ${OpenCV_INCLUDE_DIRS})
add_compile_definitions(COMPILEDWITHC11 BOOST_ALL_NO_LIB EIGEN_MPL2_ONLY)

# 全部镜像头文件使用相同Android扩展，避免混用两种类布局；不写上游源文件。
file(GLOB_RECURSE ORB_HEADERS RELATIVE "${ORB}/include" "${ORB}/include/*.h")
foreach(HEADER IN LISTS ORB_HEADERS)
    file(READ "${ORB}/include/${HEADER}" HEADER_TEXT)
    foreach(CLASS_NAME KeyFrame MapPoint Map Preintegrated)
        string(REPLACE "class ${CLASS_NAME}\n{" "class ${CLASS_NAME} : public aasc_slam::AndroidOwned<${CLASS_NAME}>\n{" HEADER_TEXT "${HEADER_TEXT}")
    endforeach()
    foreach(CLASS_NAME Pinhole KannalaBrandt8)
        string(REPLACE "class ${CLASS_NAME} : public GeometricCamera" "class ${CLASS_NAME} : public GeometricCamera, public aasc_slam::AndroidOwned<${CLASS_NAME}>" HEADER_TEXT "${HEADER_TEXT}")
    endforeach()
    if(HEADER STREQUAL "System.h")
        string(REPLACE "    // Input sensor" "    friend void DestroyAndroidSystem(System*);\n    // Input sensor" HEADER_TEXT "${HEADER_TEXT}")
    elseif(HEADER STREQUAL "LoopClosing.h")
        string(REPLACE "bool mnFullBAIdx;" "int mnFullBAIdx;" HEADER_TEXT "${HEADER_TEXT}")
        string(REPLACE "class Tracking;" "class System;\nclass Tracking;" HEADER_TEXT "${HEADER_TEXT}")
        string(REPLACE "class LoopClosing\n{" "class LoopClosing\n{\n    friend void DestroyAndroidSystem(System*);" HEADER_TEXT "${HEADER_TEXT}")
    elseif(HEADER STREQUAL "CameraModels/Pinhole.h")
        string(REPLACE "TwoViewReconstruction* tvr;" "TwoViewReconstruction* tvr = nullptr;" HEADER_TEXT "${HEADER_TEXT}")
    endif()
    get_filename_component(HEADER_DIR "${CMAKE_CURRENT_BINARY_DIR}/include/${HEADER}" DIRECTORY)
    file(MAKE_DIRECTORY "${HEADER_DIR}")
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/include/${HEADER}" CONTENT "#include \"${CMAKE_CURRENT_SOURCE_DIR}/android-ownership.h\"\n${HEADER_TEXT}")
endforeach()
include_directories(BEFORE "${CMAKE_CURRENT_BINARY_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}/include/CameraModels")

# 独立构建Android所需库，避免上游桌面工程的march=native/示例/ROS/Pangolin。
file(GLOB DBOW_SRC "${ORB}/Thirdparty/DBoW2/DBoW2/*.cpp" "${ORB}/Thirdparty/DBoW2/DUtils/*.cpp")
add_library(aasc_dbow STATIC ${DBOW_SRC})
file(GLOB G2O_SRC "${ORB}/Thirdparty/g2o/g2o/core/*.cpp" "${ORB}/Thirdparty/g2o/g2o/types/*.cpp"
    "${ORB}/Thirdparty/g2o/g2o/stuff/*.cpp")
set(G2O_CXX_COMPILER "Android Clang")
configure_file("${ORB}/Thirdparty/g2o/config.h.in" "${ORB}/Thirdparty/g2o/config.h")
include_directories("${CMAKE_CURRENT_BINARY_DIR}" "${ORB}/Thirdparty/g2o")
add_library(aasc_g2o STATIC ${G2O_SRC})
file(GLOB BOOST_SRC "${BOOST}/libs/serialization/src/*.cpp")
add_library(aasc_serialization STATIC ${BOOST_SRC})
file(GLOB ORB_SRC "${ORB}/src/*.cc" "${ORB}/src/CameraModels/*.cpp" "${ORB}/src/*.cpp")
list(FILTER ORB_SRC EXCLUDE REGEX "/(Viewer|MapDrawer|System)\\.cc$")
# Android输入错误必须回到桥错误事件，不能让上游exit直接结束APK进程。
set(PATCHED_ORB_SRC)
foreach(SOURCE IN LISTS ORB_SRC)
    file(READ "${SOURCE}" SOURCE_TEXT)
    if(SOURCE_TEXT MATCHES "exit\\(" OR SOURCE MATCHES "/(LoopClosing|Tracking)\\.cc$")
        string(REGEX REPLACE "exit\\([-0-9]+\\);" "throw std::runtime_error(\"ORB-SLAM3 configuration error\");" SOURCE_TEXT "${SOURCE_TEXT}")
        if(SOURCE MATCHES "/LoopClosing\\.cc$")
            # 上游detach后旧GBA仍会访问地图；取消时先解锁再join，杜绝悬挂优化任务。
            string(REPLACE "mpThreadGBA->detach();" "lock.unlock(); if(mpThreadGBA->joinable()) mpThreadGBA->join();" SOURCE_TEXT "${SOURCE_TEXT}")
            string(REPLACE "delete mpThreadGBA;" "delete mpThreadGBA; mpThreadGBA = nullptr;" SOURCE_TEXT "${SOURCE_TEXT}")
            string(REPLACE "mpThreadGBA = new thread" "if(mpThreadGBA) { if(mpThreadGBA->joinable()) mpThreadGBA->join(); delete mpThreadGBA; mpThreadGBA = nullptr; }\n        mpThreadGBA = new thread" SOURCE_TEXT "${SOURCE_TEXT}")
        endif()
        if(SOURCE MATCHES "/Tracking\\.cc$")
            string(REPLACE "//f_track_stats.close();" "delete mpORBextractorLeft;\n    if(mSensor == System::MONOCULAR || mSensor == System::IMU_MONOCULAR) delete mpIniORBextractor;\n    if(mSensor == System::STEREO || mSensor == System::IMU_STEREO) delete mpORBextractorRight;\n    delete mpImuCalib;" SOURCE_TEXT "${SOURCE_TEXT}")
            # 本APK始终使用新参数加载器；单目不读取Settings中未配置的IMU字段。
            string(REPLACE "//IMU parameters\n    Sophus::SE3f Tbc" "mnFramesToResetIMU = mMaxFrames; mInsertKFsLost = false; mImuFreq = 0; mImuPer = 0.001;\n    if(mSensor == System::MONOCULAR) { mpImuCalib = nullptr; mpImuPreintegratedFromLastKF = nullptr; return; }\n    //IMU parameters\n    Sophus::SE3f Tbc" SOURCE_TEXT "${SOURCE_TEXT}")
            string(REPLACE "Frame* pF = new Frame(mCurrentFrame);" "/* Android: 上游此临时Frame没有保存到任何位置，省略无效分配。 */" SOURCE_TEXT "${SOURCE_TEXT}")
            string(REPLACE "pF->mpPrevFrame = new Frame(mLastFrame);" "" SOURCE_TEXT "${SOURCE_TEXT}")
            string(REPLACE "pF->mpImuPreintegratedFrame = new IMU::Preintegrated(mCurrentFrame.mpImuPreintegratedFrame);" "" SOURCE_TEXT "${SOURCE_TEXT}")
            string(REPLACE "vector<MLPnPsolver*> vpMLPnPsolvers;" "vector<std::unique_ptr<MLPnPsolver>> vpMLPnPsolvers;" SOURCE_TEXT "${SOURCE_TEXT}")
            string(REPLACE "vpMLPnPsolvers[i] = pSolver;" "vpMLPnPsolvers[i].reset(pSolver);" SOURCE_TEXT "${SOURCE_TEXT}")
            string(REPLACE "MLPnPsolver* pSolver = vpMLPnPsolvers[i];" "MLPnPsolver* pSolver = vpMLPnPsolvers[i].get();" SOURCE_TEXT "${SOURCE_TEXT}")
        endif()
        get_filename_component(SOURCE_NAME "${SOURCE}" NAME)
        set(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/${SOURCE_NAME}")
        file(CONFIGURE OUTPUT "${SOURCE}" CONTENT "#include <stdexcept>\n#include <memory>\n${SOURCE_TEXT}")
    endif()
    list(APPEND PATCHED_ORB_SRC "${SOURCE}")
endforeach()
# 上游Shutdown未等待线程退出；Android停止/重启必须先完成线程回收再释放会话。
file(READ "${ORB}/src/System.cc" SYSTEM_SOURCE)
string(REPLACE "#include \"System.h\"" "#include \"${CMAKE_CURRENT_BINARY_DIR}/include/System.h\"" SYSTEM_SOURCE "${SYSTEM_SOURCE}")
string(REGEX REPLACE "exit\\([-0-9]+\\);" "throw std::runtime_error(\"ORB-SLAM3 initialization error\");" SYSTEM_SOURCE "${SYSTEM_SOURCE}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/System.cc" CONTENT "${SYSTEM_SOURCE}")

add_library(aasc_mmd_slam SHARED ${PATCHED_ORB_SRC} "${CMAKE_CURRENT_BINARY_DIR}/System.cc" headless-viewer.cc native-slam.cc)
target_link_libraries(aasc_mmd_slam aasc_dbow aasc_g2o aasc_serialization ${OpenCV_LIBS} android log)
target_compile_options(aasc_mmd_slam PRIVATE -O2)
